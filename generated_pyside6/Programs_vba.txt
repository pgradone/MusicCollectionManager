Private Sub SongCombo_AfterUpdate()
On Error GoTo Err_SongCombo_AfterUpdate

    Dim Criteria As String, ArtistsOfSong As String, ArtistCriteria As String
    Dim RecordCriteria As String, MyRecords As Recordset
    'dim MyContain As Recordset,
    Dim MyDb As Database, MySchedule As Recordset, RecordsOfSong As String
    Dim MySongs As Recordset, MyArtists As Recordset, MySing As Recordset
    Dim MyWS As Workspace, Myfile As String
    'Myfile = Right(CurrentDb.TableDefs(1).Connect, Len(CurrentDb.TableDefs(1).Connect) - 10)
    'Myfile = "MusiDB.mdb"
    Set MyWS = DBEngine.Workspaces(0)
    'Set MyDB = MyWS.OpenDatabase(Myfile)
    Set MyDb = CurrentDb
    Criteria = "[SongID] = " & Me![SongCombo]
    Set MySchedule = MyDb.OpenRecordset("Schedule", DB_OPEN_TABLE)
    Set MySongs = MyDb.OpenRecordset("Songs", DB_OPEN_DYNASET)
    Set MyArtists = MyDb.OpenRecordset("Artists", DB_OPEN_DYNASET)
    Set MySing = MyDb.OpenRecordset("Sing", DB_OPEN_DYNASET)
    Set MyContain = MyDb.OpenRecordset("Contain", DB_OPEN_DYNASET)
    Set MyRecords = MyDb.OpenRecordset("Records", DB_OPEN_DYNASET)

    MySongs.FindFirst Criteria
    MySchedule.Index = "PrimaryKey"
    MySchedule.AddNew
    MySchedule("ProgramID") = Me![ProgramID]   ' **** Fill Schedule IDs
    MySchedule("Position") = Me![PositionSchedule]
    MySing.FindFirst Criteria
    Do Until MySing.NoMatch  ' ********* Get all Artists singing song
       ArtistCriteria = "[ArtistID] = " & MySing("ArtistID")
       MyArtists.FindFirst ArtistCriteria
       If MyArtists("Name") <> "" Then
         ArtistsOfSong = ArtistsOfSong + " " + MyArtists("Name")
       End If
       If MyArtists("Surname") <> "" Then
         ArtistsOfSong = ArtistsOfSong + " " + MyArtists("Surname")
       End If
       MySing.FindNext Criteria
    Loop
    'MyContain.FindFirst Criteria
    'Do Until MyContain.NoMatch  '  ******** Get all Records containing
    '   RecordCriteria = "[RecordID] = " & MyContain("RecordID")
    '   MyRecords.FindFirst RecordCriteria
    '   If MyRecords("Title") <> "" Then
    '     RecordsOfSong = RecordsOfSong + "  " + MyRecords("Title")
    '   End If
    '   If MyRecords("Support") <> "" Then
    '     RecordsOfSong = RecordsOfSong + " " + MyRecords("Support")
    '   End If
    '   If MyContain("Position") <> "" Then
    '     RecordsOfSong = RecordsOfSong + " :" + MyContain("Position")
    '   End If
    '   MyContain.FindNext Criteria
    'Loop
    RecordsOfSong = Me.SongCombo.Column(3)
    ' ********* Fill Fields with calculated contents
    MySchedule("SongID") = Me![SongCombo]
    ArtistsOfSong = Trim$(ArtistsOfSong)
    MySchedule("Song_Artist") = Left$(MySongs("Title") + " * " + ArtistsOfSong, 79)
    MySchedule("Record") = Left$(Trim$(RecordsOfSong), 79)
    MySchedule("BPM") = MySongs("BPM")
    MySchedule("Year") = MySongs("Year")
    MySchedule.Update
    MyRecords.Close
    MyContain.Close
    MySchedule.Close
    MySongs.Close
    MySing.Close
    MyArtists.Close
    Me!PositionSchedule = Me!PositionSchedule + 1
    Me.Refresh

Exit_SongCombo_AfterUpdate:
    Exit Sub

Err_SongCombo_AfterUpdate:
    MsgBox Err.Description
    'MsgBox Error$ & " enter program ID!"
    Resume Exit_SongCombo_AfterUpdate

End Sub
