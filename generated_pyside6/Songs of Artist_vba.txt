Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = True
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Compare Database   'Use database order for string comparisons

Private Sub Button23_Click()

    Dim MyDb As Database, MyTable As Recordset, MyQuery As QueryDef
    Set MyDb = DBEngine.Workspaces(0).Databases(0)
    Set MyTable = MyDb.OpenRecordset("Sing", DB_OPEN_TABLE)

    MyTable.Index = "PrimaryKey"
    MyTable.Seek "=", Forms!Employees![EmployeeID], Me![CompanyID]

    If Not MyTable.NoMatch Then
        MyTable.Delete
    End If
    MyTable.Close
    Forms.Artistss.Refresh

End Sub

Private Sub ButtonRemoveSong_Click()

  RelationsMgt.RemoveFromButton "Sing", Forms!Artists![ArtistID], Me![SongID], "Surname"

End Sub

Private Sub RecordID_DblClick(Cancel As Integer)

    Dim FormName As String
    Dim LinkCriteria As String

    FormName = "Records"
    If Me!RecordID <> "" Then
        LinkCriteria = "[RecordID]=" & Screen.ActiveControl
        DoCmd.OpenForm FormName, , , LinkCriteria
    End If

End Sub

Private Sub SongCombo_AfterUpdate()

     RelationsMgt.AddFromCombo Forms!Artists![ArtistID], "Surname", Me!SongCombo, "ArtistID", "SongID", "Artists", "Sing"

End Sub

Private Sub SongID_DblClick(Cancel As Integer)

    Dim GotoCriteria As String, MyForm As String
    Dim MyKey As String, MyFirstControl As String

    If Screen.ActiveControl <> "" Then
      MyForm = "Songs"
      MyKey = "SongID"
      MyFirstControl = "Title"

      GotoCriteria = Screen.ActiveControl
      DoCmd.OpenForm MyForm
      DoCmd.GoToControl MyKey
      DoCmd.FindRecord GotoCriteria
      DoCmd.GoToControl MyFirstControl
    End If

End Sub
