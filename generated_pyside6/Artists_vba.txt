Private Sub ArtistCombo_AfterUpdate()

   Dim Criteria As String     ' This is the argument to the FindFirst method.
   Dim MyRS As Recordset      ' Recordset used to search.
   Dim ActiveName As Integer  ' The name of the ActiveName to search for.

   Set MyRS = Me.RecordsetClone

   ' Build the criteria.
   ActiveName = Screen.ActiveControl
   Criteria = "[ArtistID]=" & ActiveName

   ' Perform the search.
   MyRS.FindFirst Criteria

   If MyRS.NoMatch Then
      MsgBox "Not Found, Creating new record: " & ActiveName
      DoCmd.GoToRecord , , A_NEWREC
      Me![ArtistID] = Me!ArtistCombo
      Me.Refresh
   Else
      ' Synchronize the form's record to the dynaset's record.
      Me.Bookmark = MyRS.Bookmark
   End If

   Me!ArtistCombo = ""

End Sub

Function ArtistDuplicates() As Boolean
Dim qdf As QueryDef, rst As Recordset, sqlTxT As String
On Error Resume Next
sqlTxT = "SELECT Count([Artists].[Name] & [Artists].[Surname]) AS Duplications, " & _
"[Artists].[Name] & ' ' & [Artists].[Surname] AS DuplicatedArtist " & _
"FROM Artists WHERE ([Artists].[Name] & [Artists].[Surname])= '" & Trim(Me!Name) & Trim(Me!Surname) & "' " & _
"GROUP BY Artists.Name & ' ' & Artists.Surname " & _
"HAVING (((Count([Artists].[Name] & [Artists].[Surname]))>1)) "
Set rst = CurrentDb.OpenRecordset(sqlTxT)
With rst
    If .RecordCount >= 1 Then
        MsgBox "the artist " & "" & "already exists", vbOKOnly, "Duplicate Artist Warning"
        ArtistDuplicates = True
    End If
    .Close
End With
End Function
