Private Sub Field37_AfterUpdate()

    Dim MyDb As Database, MyTable As Recordset
    Set MyDb = DBEngine.Workspaces(0).Databases(0)
    Set MyTable = MyDb.OpenRecordset("Contain", DB_OPEN_TABLE)
    MyTable.Index = "PrimaryKey"
    MyTable.AddNew
    MyTable("SongID") = Me![SongCombo]
    MyTable("RecordID") = Me![RecordID]
    MyTable.Update
    MyTable.Close
    Me.Refresh
    Me!SongCombo = ""

End Sub

Private Sub SongCombo_AfterUpdate()

   Dim Criteria As String     ' This is the argument to the FindFirst method.
   Dim MyRS As Recordset      ' Recordset used to search.
   Dim ActiveName As String  ' The name of the ActiveName to search for.
   Dim Prompt As String, Message As String, CRLF As String
   Const MB_ICONQUESTION = 32
   Const YES = 6
   Const YES_NO = 4
   CRLF = Chr$(13)

   Prompt = "Create new one?"
   Set MyRS = Me.RecordsetClone

   ' Build the criteria.
   ActiveName = Screen.ActiveControl
   Criteria = "[SongID] = " & ActiveName
   ' Perform the search.
   MyRS.FindFirst Criteria

   If MyRS.NoMatch Then
      Message = ActiveName & " not found" & CRLF
      If MsgBox(Message & Prompt, MB_ICONQUESTION & YES_NO, "Create new Song") = YES Then
         DoCmd.GoToRecord , , A_NEWREC
         Me![SongID] = Me!SongCombo
         Me.Refresh
      End If
   Else
      ' Synchronize the form's record to the dynaset's record.
      Me.Bookmark = MyRS.Bookmark
   End If

   Me!SongCombo = ""

End Sub
