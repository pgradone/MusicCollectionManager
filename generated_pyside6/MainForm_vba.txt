Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = True
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Compare Database   'Use database order for string comparisons

Private Sub Button_OpenQuery_Rec_Click()
On Error GoTo Err_Button_OpenQuery_Rec_Click

    Dim QueryName As String
    Dim LinkCriteria As String

    QueryName = "Records_Songs_Artists"
    DoCmd.OpenQuery QueryName

Exit_Button_OpenQuery_Rec_Click:
    Exit Sub

Err_Button_OpenQuery_Rec_Click:
    MsgBox Error$
    Resume Exit_Button_OpenQuery_Rec_Click

End Sub

Private Sub ButtonOpenPrograms_Click()
On Error GoTo Err_ButtonOpenPrograms_Click

    Dim DocName As String
    Dim LinkCriteria As String

    DocName = "Programs"
    DoCmd.OpenForm DocName, , , LinkCriteria
    DoCmd.GoToRecord , "", acLast
    DoCmd.MoveSize 0, 0

Exit_ButtonOpenPrograms_Click:
    Exit Sub

Err_ButtonOpenPrograms_Click:
    MsgBox Error$
    Resume Exit_ButtonOpenPrograms_Click

End Sub
